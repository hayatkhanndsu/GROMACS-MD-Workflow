#!/usr/bin/env python3
import os
import numpy as np
import matplotlib.pyplot as plt
import MDAnalysis as mda

# === Professional Font (Safe on HPC) ===
plt.rcParams["font.family"] = "serif"
plt.rcParams["font.serif"] = ["Times", "DejaVu Serif"]

# === Create output folder ===
os.makedirs("Combined_Plots", exist_ok=True)

# === Load CA-only topology and trajectory ===
u = mda.Universe("CA_structure.pdb", "md_CA_fit_20_100.xtc")
ca = u.select_atoms("name CA")
n = ca.n_atoms

print(f"Number of CA atoms: {n}")

# === Collect positions ===
positions = []
for ts in u.trajectory:
    positions.append(ca.positions.copy())

pos = np.array(positions)

# === Compute fluctuations ===
mean_pos = pos.mean(axis=0)
fluct = pos - mean_pos

fi2 = np.mean(np.sum(fluct**2, axis=2), axis=0)
den = np.sqrt(np.outer(fi2, fi2)) + 1e-12

dccm = np.zeros((n, n))

for i in range(n):
    num = np.mean(np.sum(fluct[:, i, None, :] * fluct, axis=2), axis=0)
    dccm[i, :] = num / den[i, :]

# === Save matrix ===
np.savetxt("DCCM_Compound-9_CA.dat", dccm, fmt="%.6f")

# === Plot ===
plt.figure(figsize=(9, 8))

im = plt.imshow(
    dccm,
    cmap="bwr",
    vmin=-1,
    vmax=1,
    origin="lower"
)

cbar = plt.colorbar(im)
cbar.set_label("Correlation", fontsize=16, fontweight="bold")

plt.title(
    "DCCM – Compound 9 (Cα)",
    fontsize=20,
    fontweight="bold",
    pad=15
)

plt.xlabel("Residue Index", fontsize=18, fontweight="bold")
plt.ylabel("Residue Index", fontsize=18, fontweight="bold")

plt.grid(False)
plt.tick_params(direction="out", length=6, width=3, labelsize=14)

ax = plt.gca()
for spine in ax.spines.values():
    spine.set_linewidth(3)

plt.tight_layout()

filename = "DCCM_Compound-9_CA.png"
plt.savefig(os.path.join("Combined_Plots", filename), dpi=400)
plt.show()

print(f"✅ Saved: Combined_Plots/{filename}")
